Además, el propio Excel separa “Planificadas”, “Realizadas” y el porcentaje de cumplimiento. -- ============================================================ -- CAPACITACIONES - DUPLICAR NORTE UAM -- Estructura para carga mensual desde Excel -- ============================================================ -- ============================================================ -- 1. HISTORIAL DE CARGAS DE EXCEL -- ============================================================ CREATE TABLE IF NOT EXISTS public.capacitaciones_cargas ( id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, archivo_nombre TEXT NOT NULL, fecha_carga TIMESTAMPTZ NOT NULL DEFAULT NOW(), usuario TEXT, filas_detectadas INTEGER DEFAULT 0, eventos_detectados INTEGER DEFAULT 0 ); -- ============================================================ -- 2. REGISTRO DE CAPACITACIONES -- -- Cada fila representa una capacitación en un mes determinado. -- -- P = Planificada -- R = Planificada realizada -- RFF = Planificada realizada fuera de fecha -- RNP = Realizada no programada -- ============================================================ CREATE TABLE IF NOT EXISTS public.capacitaciones_registro ( id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, carga_id BIGINT NOT NULL REFERENCES public.capacitaciones_cargas(id) ON DELETE CASCADE, item INTEGER, tema TEXT NOT NULL, especialidades TEXT[] DEFAULT '{}', anio INTEGER NOT NULL, mes INTEGER NOT NULL CHECK (mes BETWEEN 1 AND 12), estado TEXT NOT NULL CHECK (estado IN ('P','R','RFF','RNP')), created_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); -- Evita duplicados dentro de una misma carga CREATE UNIQUE INDEX IF NOT EXISTS capacitaciones_registro_unico ON public.capacitaciones_registro ( carga_id, item, anio, mes, estado ); -- Índices para consultas rápidas CREATE INDEX IF NOT EXISTS idx_capacitaciones_registro_carga ON public.capacitaciones_registro(carga_id); CREATE INDEX IF NOT EXISTS idx_capacitaciones_registro_fecha ON public.capacitaciones_registro(anio, mes); CREATE INDEX IF NOT EXISTS idx_capacitaciones_registro_estado ON public.capacitaciones_registro(estado); -- ============================================================ -- 3. SEGURIDAD -- ============================================================ ALTER TABLE public.capacitaciones_cargas ENABLE ROW LEVEL SECURITY; ALTER TABLE public.capacitaciones_registro ENABLE ROW LEVEL SECURITY; -- Sacamos políticas anteriores si existieran DROP POLICY IF EXISTS "admin capacitaciones cargas select" ON public.capacitaciones_cargas; DROP POLICY IF EXISTS "admin capacitaciones cargas insert" ON public.capacitaciones_cargas; DROP POLICY IF EXISTS "admin capacitaciones cargas update" ON public.capacitaciones_cargas; DROP POLICY IF EXISTS "admin capacitaciones cargas delete" ON public.capacitaciones_cargas; DROP POLICY IF EXISTS "admin capacitaciones registro select" ON public.capacitaciones_registro; DROP POLICY IF EXISTS "admin capacitaciones registro insert" ON public.capacitaciones_registro; DROP POLICY IF EXISTS "admin capacitaciones registro update" ON public.capacitaciones_registro; DROP POLICY IF EXISTS "admin capacitaciones registro delete" ON public.capacitaciones_registro; -- ============================================================ -- 4. POLÍTICAS PARA ADMINISTRADORES -- -- Usa la misma función es_admin_calidad() que ya tenemos -- para Documentación / Blister / Listado Base. -- ============================================================ CREATE POLICY "admin capacitaciones cargas select" ON public.capacitaciones_cargas FOR SELECT TO authenticated USING ( public.es_admin_calidad() ); CREATE POLICY "admin capacitaciones cargas insert" ON public.capacitaciones_cargas FOR INSERT TO authenticated WITH CHECK ( public.es_admin_calidad() ); CREATE POLICY "admin capacitaciones cargas update" ON public.capacitaciones_cargas FOR UPDATE TO authenticated USING ( public.es_admin_calidad() ) WITH CHECK ( public.es_admin_calidad() ); CREATE POLICY "admin capacitaciones cargas delete" ON public.capacitaciones_cargas FOR DELETE TO authenticated USING ( public.es_admin_calidad() ); CREATE POLICY "admin capacitaciones registro select" ON public.capacitaciones_registro FOR SELECT TO authenticated USING ( public.es_admin_calidad() ); CREATE POLICY "admin capacitaciones registro insert" ON public.capacitaciones_registro FOR INSERT TO authenticated WITH CHECK ( public.es_admin_calidad() ); CREATE POLICY "admin capacitaciones registro update" ON public.capacitaciones_registro FOR UPDATE TO authenticated USING ( public.es_admin_calidad() ) WITH CHECK ( public.es_admin_calidad() ); CREATE POLICY "admin capacitaciones registro delete" ON public.capacitaciones_registro FOR DELETE TO authenticated USING ( public.es_admin_calidad() ); -- ============================================================ -- 5. PERMISOS DE LAS TABLAS PRIVADAS -- ============================================================ REVOKE ALL ON public.capacitaciones_cargas FROM anon; REVOKE ALL ON public.capacitaciones_registro FROM anon; GRANT SELECT, INSERT, UPDATE, DELETE ON public.capacitaciones_cargas TO authenticated; GRANT SELECT, INSERT, UPDATE, DELETE ON public.capacitaciones_registro TO authenticated; -- ============================================================ -- 6. VISTA PÚBLICA PARA EL DASHBOARD DEL CLIENTE -- -- IMPORTANTE: -- Solo muestra cantidades agregadas. -- NO muestra: -- - personal a capacitar -- - instructor -- - usuarios -- - datos internos de carga -- ============================================================ DROP VIEW IF EXISTS public.v_capacitaciones_dashboard; CREATE VIEW public.v_capacitaciones_dashboard AS WITH ultima_carga AS ( SELECT id FROM public.capacitaciones_cargas ORDER BY fecha_carga DESC, id DESC LIMIT 1 ), datos AS ( SELECT cr.* FROM public.capacitaciones_registro cr INNER JOIN ultima_carga uc ON uc.id = cr.carga_id ), meses AS ( SELECT anio, mes, COUNT(*) FILTER ( WHERE estado IN ('P','R','RFF') )::INTEGER AS planificadas, COUNT(*) FILTER ( WHERE estado IN ('R','RFF') )::INTEGER AS realizadas_planificadas, COUNT(*) FILTER ( WHERE estado = 'RNP' )::INTEGER AS realizadas_no_programadas, COUNT(*) FILTER ( WHERE estado IN ('R','RFF','RNP') )::INTEGER AS realizadas_totales, COUNT(*) FILTER ( WHERE estado = 'RFF' )::INTEGER AS realizadas_fuera_fecha FROM datos GROUP BY anio, mes ) SELECT anio, mes, CASE mes WHEN 1 THEN 'Enero' WHEN 2 THEN 'Febrero' WHEN 3 THEN 'Marzo' WHEN 4 THEN 'Abril' WHEN 5 THEN 'Mayo' WHEN 6 THEN 'Junio' WHEN 7 THEN 'Julio' WHEN 8 THEN 'Agosto' WHEN 9 THEN 'Septiembre' WHEN 10 THEN 'Octubre' WHEN 11 THEN 'Noviembre' WHEN 12 THEN 'Diciembre' END AS nombre_mes, planificadas, realizadas_planificadas, realizadas_no_programadas, realizadas_totales, realizadas_fuera_fecha, CASE WHEN planificadas = 0 THEN 0 ELSE ROUND( ( realizadas_planificadas::NUMERIC / planificadas::NUMERIC ) * 100, 1 ) END AS cumplimiento_porcentaje FROM meses ORDER BY anio, mes; -- ============================================================ -- 7. PERMISO PÚBLICO SOLO PARA LA VISTA -- ============================================================ GRANT SELECT ON public.v_capacitaciones_dashboard TO anon, authenticated; -- ============================================================ -- FIN -- ============================================================ Qué va a calcular Para el cumplimiento mensual voy a usar: Planificadas = P + R + RFF Realizadas planificadas = R + RFF RNP se muestra aparte, porque fue una capacitación realizada pero no estaba programada.

function matrizPintarEspecialidades(){ const nombres = [ "Civil", "Piping", "Estructuras", "Mecánica", "Electricidad", "Instrumentación" ]; const cont = document.getElementById("matrizEspecialidades"); if(!cont) return; const normalizar = valor => String(valor || "") .normalize("NFD") .replace(/[\u0300-\u036f]/g,"") .trim() .toUpperCase(); cont.innerHTML = nombres.map(nombre => { const filas = matrizFilas.filter(f => { const valor = normalizar(f.especialidad); if(nombre === "Estructuras"){ return valor === "EST" || valor.includes("ESTRUCT"); } return valor === normalizar(nombre); }); const t = matrizContar(filas); const elaborados = t.cliente + t.cerrado; const elab = matrizPct(elaborados,t.aplicables); const cerr = matrizPct(t.cerrado,t.aplicables); return `

${matrizEsc(nombre)}

Registros: ${t.aplicables.toLocaleString("es-AR")}
Avance elaborado
${elaborados.toLocaleString("es-AR")} registros (${elab.toFixed(1)}%)
Avance cerrado
${t.cerrado.toLocaleString("es-AR")} registros (${cerr.toFixed(1)}%)
`; }).join(""); }
